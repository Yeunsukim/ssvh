43	-- SSV 병원 전산 실습 — 자료 규칙표에서 자동으로 만든 테이블 생성문 (PostgreSQL 기준) -- 실습 기록(JSON)의 각 묶음이 아래 테이블 한 개와 #로 대응합니다. CREATE TABLE doctors (id VARCHAR(#) PRIMARY KEY, name TEXT NOT NULL, dept TEXT); -- 의사 명단 (마스터) CREATE TABLE master_items (code VARCHAR(#) PRIMARY KEY, cat VARCHAR(#) NOT NULL, name TEXT NOT NULL, price NUMERIC, detail JSON); -- 처방 항목 마스터 CREATE TABLE patients ( -- 환자 "id" VARCHAR(#) PRIMARY KEY, -- PK "mrn" VARCHAR(#) NOT NULL UNIQUE, -- 등록번호(고유) "name" TEXT NOT NULL, -- 이름 "sex" TEXT NOT NULL, -- 성별 M/F "birth" TEXT NOT NULL, -- 생년월일 YYYY-MM-DD "blood" TEXT, -- 혈액형 "height" NUMERIC, -- 키 cm "weight" NUMERIC, -- 몸무게 kg "ins" NUMERIC, -- 보험 종류 "phone" TEXT, -- 휴대전화 "address" TEXT, -- 주소 "guardian" TEXT, -- 보호자 "allergies" JSON, -- JSON 알레르기 목록 "flags" JSON, -- JSON 경고(격리·DNR 등) "hx" TEXT, -- 과거력 "erp" JSON, -- JSON 원무 정보 "crm" JSON, -- JSON 고객관리 정보 "refConsent" JSON, -- JSON 진료정보 제공 동의 "scenario" JSON, -- JSON 실습 사례값 "extra" JSON -- 규칙표에 없는 추가 칸 ); CREATE TABLE encounters ( -- 진료(내원·입원) "id" VARCHAR(#) PRIMARY KEY, -- PK "patientId" VARCHAR(#) NOT NULL REFERENCES patients("id"), -- FK 환자 "type" VARCHAR(#) NOT NULL, -- 외래 OPD/입원 IPD/응급 ER "dept" TEXT NOT NULL, -- 진료과 "doctorId" VARCHAR(#) NOT NULL REFERENCES doctors(id), -- FK 의사 "status" VARCHAR(#) NOT NULL, -- 상태 "at" BIGINT NOT NULL, -- 시작 시각 "endAt" BIGINT, -- 종료(퇴원) 시각 "ward" TEXT, -- 병동 "room" TEXT, -- 병실 "bed" TEXT, -- 침상 "ktas" TEXT, -- 응급 중증도 "cc" TEXT, -- 주호소 "dx" JSON, -- JSON 진단 목록(KCD) "diet" TEXT, -- 식이 "dc" JSON, -- JSON 퇴원 정보 "coding" JSON, -- JSON 코딩(의료정보) "mi" JSON, -- JSON 퇴원분석 "moves" JSON, -- JSON 전동 이력 "billed" TEXT, -- 수납 완료 여부 "extra" JSON -- 규칙표에 없는 추가 칸 ); CREATE TABLE orders ( -- 처방 "id" VARCHAR(#) PRIMARY KEY, -- PK "encId" VARCHAR(#) NOT NULL REFERENCES encounters("id"), -- FK 진료 "patientId" VARCHAR(#) NOT NULL REFERENCES patients("id"), -- FK 환자 "encType" TEXT, -- 진료 구분 "cat" VARCHAR(#) NOT NULL, -- 처방 분류 "code" VARCHAR(#) NOT NULL REFERENCES master_items(code), -- FK 처방 항목(마스터) "name" TEXT NOT NULL, -- 처방명 "dose" NUMERIC, -- 용량 "unit" TEXT, -- 단위 "route" TEXT, -- 경로 "freq" TEXT, -- 횟수 "days" NUMERIC, -- 일수 "qty" NUMERIC, -- 총량 "prn" TEXT, -- 필요시 "stat" TEXT, -- 응급 "outside" TEXT, -- 원외처방 "status" VARCHAR(#) NOT NULL, -- 진행 상태 "at" BIGINT NOT NULL, -- 처방 시각 "by" TEXT, -- 처방자 "doctorId" VARCHAR(#) REFERENCES doctors(id), -- FK 의사 "ward" TEXT, -- 병동 "hist" JSON, -- JSON 단계 이력 "result" JSON, -- JSON 결과 "specimenAt" BIGINT, -- 채혈 시각 "shotAt" BIGINT, -- 촬영 시각 "arrivedAt" BIGINT, -- 병동 도착 시각 "op" JSON, -- JSON 수술 "cv" JSON, -- JSON 심혈관 시술 "en" JSON, -- JSON 내시경 "adj" JSON, -- JSON 심사 조정 "extra" JSON -- 규칙표에 없는 추가 칸 ); CREATE TABLE notes ( -- 의사 기록 "id" VARCHAR(#) PRIMARY KEY, -- PK "encId" VARCHAR(#) NOT NULL REFERENCES encounters("id"), -- FK 진료 "patientId" VARCHAR(#) NOT NULL REFERENCES patients("id"), -- FK 환자 "type" VARCHAR(#) NOT NULL, -- 기록 종류 "at" BIGINT NOT NULL, -- 작성 시각 "by" TEXT, -- 작성자 "signed" TEXT, -- 전자서명 "content" JSON, -- JSON 내용 "extra" JSON -- 규칙표에 없는 추가 칸 ); CREATE TABLE vitals ( -- 활력징후 "id" VARCHAR(#) PRIMARY KEY, -- PK "patientId" VARCHAR(#) NOT NULL REFERENCES patients("id"), -- FK 환자 "encId" VARCHAR(#) REFERENCES encounters("id"), -- FK 진료 "at" BIGINT NOT NULL, -- 측정 시각 "sbp" NUMERIC, -- 수축기 "dbp" NUMERIC, -- 이완기 "hr" NUMERIC, -- 맥박 "rr" NUMERIC, -- 호흡 "bt" TEXT, -- 체온 "spo#" NUMERIC, -- 산소포화도 "pain" NUMERIC, -- 통증 "extra" JSON -- 규칙표에 없는 추가 칸 ); CREATE TABLE io ( -- 섭취·배설 "id" VARCHAR(#) PRIMARY KEY, -- PK "patientId" VARCHAR(#) NOT NULL REFERENCES patients("id"), -- FK 환자 "at" BIGINT NOT NULL, -- 시각 "kind" VARCHAR(#), -- 섭취/배설 "type" VARCHAR(#), -- 종류 "amount" NUMERIC, -- 양 mL "extra" JSON -- 규칙표에 없는 추가 칸 );
